  %% ── Requirement Sources ──
  JAMA["Jama / Word Export
YAML requirement files
REQ-SYS-AEB-xxx.yaml"]
  GHERKIN["Gherkin Scenarios
Feature-engineer authored
.feature files in GitHub"]
  XLSX["Spreadsheet Import
Requirements matrix
.xlsx rows"]

  %% ── Devin Session ──
  DEVIN["Devin Session Triggered
Prompt or playbook macro"]

  %% ── Analysis ──
  ANALYZE["Requirement Analysis
Assess atomicity, testability,
and ambiguity"]

  %% ── Decision ──
  ATOMIC{"Atomic & Testable?"}

  %% ── Decomposition ──
  DECOMPOSE["Decompose
Split into atomic Gherkin
scenarios with traceability"]

  %% ── Mapping ──
  MAP["Abstraction Mapping
Map abstract steps through
action library + bench boundary"]

  %% ── Generation ──
  GEN["Test Artifact Generation
Produce executable test scripts
ECU-TEST style, no hard-coded CAN IDs"]

  %% ── Execution ──
  EXEC["Test Execution
Run against HIL bench
replay harness in demo"]

  %% ── Impact ──
  IMPACT["Impact Analysis
Check existing tests affected
by requirement changes"]

  %% ── Traceability ──
  TRACE["Traceability Report
Requirement → Scenario →
Test → Result matrix"]

  %% ── PR ──
  PR["PR with All Artifacts
Scenarios, test scripts, results,
traceability report"]

  %% ── Review ──
  REVIEW["Engineer Reviews
Test engineer approves
or requests changes"]

  %% ── Decision ──
  APPROVED{"Approved?"}
  READY["Ready for Bench Execution
ISO 26262 traceability maintained"]
  ITERATE["Iterate
Refine scenarios or tests"]

  %% ── Connections ──
  JAMA --> DEVIN
  GHERKIN --> DEVIN
  XLSX --> DEVIN
  DEVIN --> ANALYZE
  ANALYZE --> ATOMIC
  ATOMIC -- "Yes" --> MAP
  ATOMIC -- "No" --> DECOMPOSE
  DECOMPOSE --> MAP
  MAP --> GEN
  GEN --> EXEC
  EXEC --> IMPACT
  IMPACT --> TRACE
  TRACE --> PR
  PR --> REVIEW
  REVIEW --> APPROVED
  APPROVED -- "Yes" --> READY
  APPROVED -- "No" --> ITERATE
  ITERATE -. "refine" .-> ANALYZE
