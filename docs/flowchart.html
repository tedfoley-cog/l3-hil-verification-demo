flowchart TD
  %% ── Requirement Sources ──
  JAMA["Jama / Word Export\nYAML requirement files\nREQ-SYS-AEB-xxx.yaml"]
  GHERKIN["Gherkin Scenarios\nFeature-engineer authored\n.feature files in GitHub"]
  XLSX["Spreadsheet Import\nRequirements matrix\n.xlsx rows"]

  %% ── Devin Session ──
  DEVIN["Devin Session Triggered\nPrompt or playbook macro"]

  %% ── Analysis ──
  ANALYZE["Requirement Analysis\nAtomicity, testability,\nambiguity assessment"]

  %% ── Decision ──
  ATOMIC{"Atomic & Testable?"}

  %% ── Decomposition ──
  DECOMPOSE["Decompose\nSplit into atomic Gherkin\nscenarios with traceability"]

  %% ── Mapping ──
  MAP["Abstraction Mapping\nAction library + bench\nboundary resolution"]

  %% ── Generation ──
  GEN["Test Artifact Generation\nECU-TEST executable scripts\nNo hard-coded CAN IDs"]

  %% ── Execution ──
  EXEC["Test Execution\nRun against HIL bench\nReplay harness in demo"]

  %% ── Impact ──
  IMPACT["Impact Analysis\nIdentify existing tests\naffected by changes"]

  %% ── Traceability ──
  TRACE["Traceability Report\nReq > Scenario >\nTest > Result matrix"]

  %% ── PR ──
  PR["PR with All Artifacts\nScenarios, scripts, results,\ntraceability report"]

  %% ── Review ──
  REVIEW["Engineer Reviews\nTest engineer approves\nor requests changes"]

  %% ── Decision ──
  APPROVED{"Approved?"}
  READY["Bench Ready\nISO 26262 traceable"]
  ITERATE["Iterate\nRefine scenarios or tests"]

  %% ── Connections ──
  JAMA --> DEVIN
  GHERKIN --> DEVIN
  XLSX --> DEVIN
  DEVIN --> ANALYZE
  ANALYZE --> ATOMIC
  ATOMIC -- "Yes" --> MAP
  ATOMIC -- "No" --> DECOMPOSE
  DECOMPOSE --> MAP
  MAP --> GEN
  GEN --> EXEC
  EXEC --> IMPACT
  IMPACT --> TRACE
  TRACE --> PR
  PR --> REVIEW
  REVIEW --> APPROVED
  APPROVED -- "Yes" --> READY
  APPROVED -- "No" --> ITERATE
  ITERATE -. "refine" .-> ANALYZE
